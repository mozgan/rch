function(rch_add_unit_test target_name)
    add_executable(${target_name} ${target_name}.cpp)
    target_link_libraries(${target_name} PRIVATE rch::core GTest::gtest_main)
    gtest_discover_tests(${target_name} DISCOVERY_MODE PRE_TEST)
endfunction()
