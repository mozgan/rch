function(rch_add_property_test target_name)
    add_executable(${target_name} ${target_name}.cpp)
    target_include_directories(${target_name} PRIVATE "${CMAKE_SOURCE_DIR}")
    target_link_libraries(${target_name} PRIVATE rch::core GTest::gtest_main)
    gtest_discover_tests(${target_name} NO_PRETTY_TYPES NO_PRETTY_VALUES DISCOVERY_MODE PRE_TEST)
endfunction()

rch_add_property_test(prop_quantization_bounds)
rch_add_property_test(prop_hilbert_range)
rch_add_property_test(prop_curve_intrinsics)
rch_add_property_test(prop_sort_determinism)
rch_add_property_test(prop_permutation_invariance)
rch_add_property_test(prop_total_order)
rch_add_property_test(prop_fallback_semantics)
rch_add_property_test(test_property_generators)
